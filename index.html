Factorial (Subroutine)

AREA factorial,CODE,READONLY
EXPORT __main

__main PROC
LDR r0,=0x20000000
LDR r1,[r0,#0]
BL fact
STR r3,[r0,#4]

l b l
ENDP

fact PROC
mov r2,r1
mov r3,#1

loop
mul r3,r2,r3
sub r2,r2,#1
cmp r2,#0
bne loop

BX LR
ENDP
end

Recursive Factorial

AREA factorialrec,CODE,READONLY
EXPORT __main

__main PROC
LDR r0,=0x20000000
LDR r1,[r0,#0]
BL fact
STR r1,[r0,#4]

l b l
ENDP

fact PROC
sub sp,sp,#8
STR LR,[sp,#4]
STR r1,[sp,#0]

cmp r1,#1
bgt L1
b lpop

L1
sub r1,r1,#1
BL fact

lpop
mov r12,r1
LDR r1,[sp,#0]
LDR LR,[sp,#4]
ADD sp,sp,#8

mul r1,r12,r1
mov pc,LR

ENDP
END

STM32 Timer Delay

#include

void delay(){

RCC->APB1ENR |=(1<<0);

TIM2->PSC=7199;
TIM2->ARR=499;
TIM2->SR=0;

TIM2->CR1=1;

while((TIM2->SR & 1)==0);

TIM2->CR1=0;

}

int main(){

RCC->APB2ENR |=0xFC;
GPIOC->CRH=0x44344444;

while(1){

GPIOC->ODR^=(1<<13);
delay();

}
}

Sorting

AREA MYCODE, CODE, READONLY
        EXPORT __main
__main
        LDR R0, =ARRAY          ; source array
        LDR R6, =0x20000000     ; destination RAM
        MOV R5, #5              ; number of elements
COPY
        LDR R1, [R0], #4        ; load from ARRAY
        STR R1, [R6], #4        ; store to RAM
        SUBS R5, R5, #1
        BNE COPY

        MOV R4,#4               ; outer loop

OUTER
        LDR R6, =0x20000000     ; reset pointer
        MOV R2,#4               ; inner loop

INNER
        LDR R3,[R6]
        LDR R1,[R6,#4]
        CMP R3,R1
        BLE NOSWAP

        STR R1,[R6]
        STR R3,[R6,#4]

NOSWAP
        ADD R6,R6,#4
        SUBS R2,R2,#1
        BNE INNER

        SUBS R4,R4,#1
        BNE OUTER

STOP
        B STOP
ARRAY DCD 5,1,4,2,3
        END
